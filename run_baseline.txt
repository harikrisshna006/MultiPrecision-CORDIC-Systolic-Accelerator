# Gmail-safe copy of the paper-derived Flex-PE baseline runner.
# Rename this file to run_baseline.ps1 after downloading it, then execute it
# from a ModelSim / Questa PowerShell prompt.
$ErrorActionPreference = 'Stop'
$projectRoot = Split-Path -Parent $MyInvocation.MyCommand.Path
Set-Location -LiteralPath $projectRoot

$library = 'build_modelsim_baseline'
if (Test-Path -LiteralPath $library) {
    Remove-Item -LiteralPath $library -Recurse -Force
}

vlib $library
$dutFiles = Get-ChildItem -LiteralPath rtl -File -Filter '*.v' |
    Where-Object { $_.Name -notlike 'tb_*.v' } |
    Sort-Object Name |
    ForEach-Object { $_.FullName }

$testFiles = @(
    'rtl\tb_control_fsm.v',
    'rtl\tb_cordic_stage_iterative.v',
    'rtl\tb_cordic_stage_pipeline.v',
    'rtl\tb_flex_pe_iterative_top.v',
    'rtl\tb_flex_pe_pipelined_top.v',
    'rtl\tb_flex_pe_config_fsm.v',
    'rtl\tb_flex_pe_simd64_iterative.v',
    'rtl\tb_flex_pe_simd64_pipelined.v',
    'rtl\tb_flex_pe_simd64_array_3x3.v'
)

& vlog -work $library -stats=none $dutFiles $testFiles
if ($LASTEXITCODE -ne 0) { throw 'Compilation failed. Read the vlog output above.' }

$tests = @(
    'tb_control_fsm',
    'tb_cordic_stage_iterative',
    'tb_cordic_stage_pipeline',
    'tb_flex_pe_iterative_top',
    'tb_flex_pe_pipelined_top',
    'tb_flex_pe_config_fsm',
    'tb_flex_pe_simd64_iterative',
    'tb_flex_pe_simd64_pipelined',
    'tb_flex_pe_simd64_array_3x3'
)

$failed = @()
foreach ($test in $tests) {
    Write-Output ''
    Write-Output "===== RUNNING $test ====="
    $output = & vsim -c -quiet -lib $library $test -do 'run -all; quit -f' 2>&1
    $output | ForEach-Object { Write-Output $_ }
    if ($LASTEXITCODE -ne 0 -or ($output -join "`n") -match 'FAIL=[1-9]') {
        $failed += $test
    }
}

if ($failed.Count -gt 0) {
    throw ('Baseline verification failed: ' + ($failed -join ', '))
}

Write-Output ''
Write-Output "BASELINE VERIFICATION PASSED: $($tests.Count) testbenches."
